### 筛选逻辑

```typescript
type FilterType = 'all' | 'favorites' | 'active' | 'regular';

function filterProjects(projects: Project[], filter: FilterType, searchQuery: string): Project[] {
  let result = projects;

  // Tab 筛选
  switch (filter) {
    case 'favorites':
      result = result.filter(p => p.isFavorite);
      break;
    case 'active':
      result = result.filter(p => p.hasActiveContainer);
      break;
    case 'regular':
      result = result.filter(p => !p.hasActiveContainer);
      break;
  }

  // 搜索叠加
  if (searchQuery.trim()) {
    const q = searchQuery.toLowerCase();
    result = result.filter(p =>
      p.title.toLowerCase().includes(q) ||
      p.description.toLowerCase().includes(q)
    );
  }

  // 默认按更新时间降序
  return result.sort((a, b) =>
    new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime()
  );
}
```

### Active Tab 的绿色圆点

`● Active` Tab 前的绿色圆点是静态的（不带呼吸动画），与卡片上的脉搏灯视觉呼应但不重复动画效果。选中态时圆点颜色不变（始终 `bg-green-500`），文字跟随 Tab 选中样式变化。

### 数据结构

```typescript
interface DayActivity {
  date: string;           // "2026-02-20"
  updateCount: number;    // 当天所有项目的更新总次数
  projectIds: string[];   // 当天有更新的项目 ID 列表
}

// Mock 数据：生成最近 84 天（12 周）的随机活动数据
function generateMockHeatmapData(projects: Project[]): DayActivity[] {
  const days: DayActivity[] = [];
  const now = new Date();
  for (let i = 83; i >= 0; i--) {
    const date = new Date(now);
    date.setDate(date.getDate() - i);
    const dateStr = date.toISOString().split('T')[0];
    const updateCount = Math.random() > 0.6 ? Math.floor(Math.random() * 8) : 0;
    const activeProjects = updateCount > 0
      ? projects.slice(0, Math.min(updateCount, projects.length)).map(p => p.id)
      : [];
    days.push({ date: dateStr, updateCount, projectIds: activeProjects });
  }
  return days;
}
```

### 色阶

复用脉搏灯绿色（`#22c55e`），通过透明度区分强度：

### 组件实现

```tsx
function ActivityHeatmap({
  data,
  selectedDate,
  onSelectDate,
}: {
  data: DayActivity[];
  selectedDate: string | null;
  onSelectDate: (date: string | null) => void;
}) {
  // 按周分组，每列 = 一周（7 天）
  const weeks = useMemo(() => {
    const result: DayActivity[][] = [];
    for (let i = 0; i < data.length; i += 7) {
      result.push(data.slice(i, i + 7));
    }
    return result;
  }, [data]);

  function getCellLevel(count: number): number {
    if (count === 0) return 0;
    if (count <= 2) return 1;
    if (count <= 5) return 2;
    return 3;
  }

  return (
    <div className="mb-4">
      <div className="flex gap-[2px]">
        {/* 行标签 */}
        <div className="flex flex-col justify-between text-[10px] text-muted-foreground/50 pr-1 py-[1px]">
          <span>Mon</span>
          <span>Wed</span>
          <span>Fri</span>
        </div>
        {/* 色块网格 */}
        <div className="flex gap-[2px]">
          {weeks.map((week, wi) => (
            <div key={wi} className="flex flex-col gap-[2px]">
              {week.map((day) => (
                <div
                  key={day.date}
                  className={`heatmap-cell heatmap-cell-${getCellLevel(day.updateCount)}
                    ${day.updateCount > 0 ? 'cursor-pointer' : ''}
                    ${selectedDate === day.date ? 'ring-1 ring-white' : ''}`}
                  onClick={() => {
                    if (day.updateCount === 0) return;
                    onSelectDate(selectedDate === day.date ? null : day.date);
                  }}
                  title={`${formatHeatmapDate(day.date)}: ${day.updateCount} updates in ${day.projectIds.length} projects`}
                />
              ))}
            </div>
          ))}
        </div>
      </div>
      {/* 月份标签 */}
      <div className="flex text-[10px] text-muted-foreground/40 mt-1 ml-8">
        {/* 根据 data 中的月份变化点渲染月份标签 */}
      </div>
    </div>
  );
}
```

- 使用现有 design tokens，不硬编码颜色
