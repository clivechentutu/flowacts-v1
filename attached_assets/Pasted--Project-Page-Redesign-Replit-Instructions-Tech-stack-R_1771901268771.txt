# Project Page Redesign — Replit Instructions

> Tech stack: React + TypeScript + Tailwind CSS + Radix UI + Framer Motion
> Theme: Dark mode (use existing design tokens: bg-card, bg-background, text-foreground, text-muted-foreground, border-border)
> This is a redesign of the existing Projects page. Do NOT create a new page — modify the existing one.

---

## Overview

Redesign the Projects page from a CMS-style card layout (large cover images + title + description) to a compact, status-driven card layout. The new cards use a unified 3-line structure that adapts to different project types (URL analysis, AI conversation, document analysis, etc.).

**Key changes:**
1. Remove cover images from project cards entirely
2. Replace with a compact 3-line card: Title → Context summary → Dynamic status + Canvas count
3. Add project status grouping (Pinned / Active / Completed / Archived)
4. Add [+ New Project] button in the top bar
5. Add card context menu (right-click / ··· button)

```tsx
function DynamicStatus({ project }: { project: Project }) {
  // Priority 1: Container has new changes
  if (project.containerStatus?.overall === "new_changes") {
    return (
      <span className="text-xs font-medium text-blue-400">
        🔵 {project.containerStatus.newChangesCount} new changes · {timeAgo(project.containerStatus.lastCheckedAt)}
      </span>
    );
  }

  // Priority 2: Container failed
  if (project.containerStatus?.overall === "failed") {
    return (
      <span className="text-xs text-red-400">
        🔴 {project.containerStatus.count} failed · {timeAgo(project.containerStatus.lastCheckedAt)}
      </span>
    );
  }

  // Priority 3: Container running
  if (project.containerStatus?.overall === "running") {
    return (
      <span className="text-xs text-green-400">
        🟢 Running · updated {timeAgo(project.containerStatus.lastCheckedAt)}
      </span>
    );
  }

  // Priority 4: Container paused
  if (project.containerStatus?.overall === "paused") {
    return (
      <span className="text-xs text-yellow-400">
        🟡 Paused
      </span>
    );
  }

  // Priority 5: Completed project
  if (project.status === "completed") {
    return (
      <span className="text-xs text-muted-foreground">
        Completed {formatDate(project.completedAt)}
      </span>
    );
  }

  // Default: No container, show last updated
  return (
    <span className="text-xs text-muted-foreground">
      Updated {timeAgo(project.updatedAt)}
    </span>
  );
}
```

### Project Grouping

Group projects by status. Use this grouping logic:

```typescript
function groupProjects(projects: Project[], searchQuery: string) {
  // 1. Filter by search query
  let filtered = projects;
  if (searchQuery.trim()) {
    const q = searchQuery.toLowerCase();
    filtered = projects.filter(p =>
      p.name.toLowerCase().includes(q) ||
      (p.contextSummary?.toLowerCase().includes(q))
    );
  }

  // 2. Group
  const pinned = filtered.filter(p => p.isPinned);
  const active = filtered.filter(p => p.status === "active" && !p.isPinned);
  const completed = filtered.filter(p => p.status === "completed" && !p.isPinned);
  const archived = filtered.filter(p => p.status === "archived");

  // 3. Sort each group: projects with new_changes first, then by updatedAt desc
  const sortFn = (a: Project, b: Project) => {
    const aHasChanges = a.containerStatus?.overall === "new_changes" ? 1 : 0;
    const bHasChanges = b.containerStatus?.overall === "new_changes" ? 1 : 0;
    if (aHasChanges !== bHasChanges) return bHasChanges - aHasChanges;
    return new Date(b.updatedAt).getTime() - new Date(a.updatedAt).getTime();
  };

  return {
    pinned: pinned.sort(sortFn),
    active: active.sort(sortFn),
    completed: completed.sort(sortFn),
    archived: archived.sort(sortFn),
  };
}
```

### Group Section Component

Each group has a header label and a collapsible card grid:

```tsx
function ProjectGroup({
  label,
  icon,
  projects,
  defaultCollapsed = false,
}: {
  label: string;
  icon: string;
  projects: Project[];
  defaultCollapsed?: boolean;
}) {
  const [collapsed, setCollapsed] = useState(defaultCollapsed);

  if (projects.length === 0) return null; // Don't render empty groups

  return (
    <div className="mb-6">
      {/* Group header */}
      <button
        onClick={() => setCollapsed(!collapsed)}
        className="flex items-center gap-2 mb-3 text-xs font-medium text-muted-foreground uppercase tracking-wider hover:text-foreground transition-colors"
      >
        <span>{icon}</span>
        <span>{label}</span>
        <span className="text-muted-foreground/60">{projects.length} projects</span>
        <ChevronDown className={cn("w-3 h-3 transition-transform", collapsed && "-rotate-90")} />
      </button>

      {/* Card grid */}
      {!collapsed && (
        <div className="grid grid-cols-[repeat(auto-fill,minmax(280px,1fr))] gap-3">
          {projects.map(project => (
            <ProjectCard key={project.id} project={project} />
          ))}
        </div>
      )}
    </div>
  );
}
```

```tsx

      <DropdownMenuItem onClick={() => reopenProject(project)}>
        <RotateCcw className="w-4 h-4 mr-2" />

      <DropdownMenuItem onClick={() => archiveProject(project)}>
        <Archive className="w-4 h-4 mr-2" />

      </DropdownMenuItem>

    {project.status === "archived" && (
      <DropdownMenuItem onClick={() => unarchiveProject(project)}>
        <RotateCcw className="w-4 h-4 mr-2" />
        Unarchive
      </DropdownMenuItem>
    )}

    <DropdownMenuSeparator />

      onClick={() => confirmDelete(project)}
      className="text-red-400 focus:text-red-400"

      <Trash2 className="w-4 h-4 mr-2" />

    </DropdownMenuItem>
  </DropdownMenuContent>
```
