## Fix 2: Unified Process Timeline

### What Changes
Replace the current "Process details" (which only shows active step's process) with a unified Process Timeline that shows ALL steps' process details in a single scrollable container.

### Implementation

Replace the Process details section in DynamicExecutionCard with this new ProcessTimeline component:

```jsx
function ProcessTimeline({ steps, stepMessages }) {
  const [isExpanded, setIsExpanded] = React.useState(false);

  // Collect ALL process messages across ALL steps
  const allProcessGroups = steps
    .filter((step) => step.status === "done" || step.status === "active")
    .map((step) => {
      const msgs = stepMessages[step.id] || [];
      const processMsgs = msgs.filter((m) => {
        const level = m.messageLevel || getDefaultLevel(m);
        return level === "process";
      });
      const allSteps = processMsgs.flatMap(
        (m) => m.thoughtProcess?.steps || []
      );
      return { step, processSteps: allSteps };
    })
    .filter((group) => group.processSteps.length > 0);

  const totalProcessCount = allProcessGroups.reduce(
    (sum, g) => sum + g.processSteps.length, 0
  );

  const hasActive = allProcessGroups.some((g) =>
    g.processSteps.some((s) => s.status === "active")
  );

  // Auto-expand when there's an active process step
  const expanded = isExpanded || hasActive;

  if (totalProcessCount === 0) return null;

  const agentIcons = {
    scout: "🕵️", capturer: "📸", analyst: "📊",
    comparator: "⚖️", reporter: "📝",
  };

  return (
    <div className="border-t border-border/10">
      <button
        onClick={() => setIsExpanded(!isExpanded)}
        className="w-full flex items-center gap-2 px-4 py-2
          text-left hover:bg-muted/8 transition-colors"
      >
        <span className="text-[10px] text-muted-foreground/30
          shrink-0 w-3">
          {expanded ? "▾" : "▸"}
        </span>
        <span className="text-[11px] text-muted-foreground/40">
          Process Timeline ({totalProcessCount})
        </span>
      </button>

      {expanded && (
        <div className="px-4 pb-3 max-h-[200px] overflow-y-auto">
          <div className="space-y-3">
            {allProcessGroups.map((group) => {
              const icon =
                agentIcons[group.step.agentRole] || "🤖";
              const isActiveStep =
                group.step.status === "active";

              return (
                <div key={group.step.id}>
                  {/* Step group header */}
                  <div className="text-[10px] text-muted-foreground/40
                    font-medium tracking-wider mb-1">
                    {icon} {group.step.label}
                  </div>

                  {/* Process steps */}
                  <div className="ml-2 space-y-0.5">
                    {group.processSteps.map((ps, i) => (
                      <div
                        key={i}
                        className={`flex items-center gap-1.5
                          text-[10px] ${
                            isActiveStep
                              ? "text-muted-foreground/50"
                              : "text-muted-foreground/30"
                          }`}
                      >
                        <span
                          className={`w-1 h-1 rounded-full
                            shrink-0 ${
                              ps.status === "done"
                                ? isActiveStep
                                  ? "bg-muted-foreground/30"
                                  : "bg-muted-foreground/15"
                                : ps.status === "active"
                                ? "bg-blue-400 animate-pulse"
                                : "bg-muted-foreground/10"
                            }`}
                        />
                        <span>{ps.label}</span>
                        {ps.status === "done" && (
                          <span className="text-muted-foreground/20
                            ml-auto">✓</span>
                        )}
                      </div>
                    ))}
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      )}
    </div>
  );
}
```

function StepExpandedContent({ step, messages }) {
  const accentBorder = {
    scout: "border-blue-400/20",
    capturer: "border-purple-400/20",
    analyst: "border-orange-400/20",
    comparator: "border-green-400/20",
    reporter: "border-pink-400/20",
  };

  const guideLine = accentBorder[step.agentRole]
    || "border-border/10";

  // Separate L1 insights from L2 progress
  const resultMsgs = messages.filter((m) => {
    const level = m.messageLevel || getDefaultLevel(m);
    return level === "insight" || level === "progress";
  });

  const insightMsgs = resultMsgs.filter((m) => {
    const level = m.messageLevel || getDefaultLevel(m);
    return level === "insight";
  });

  const progressMsgs = resultMsgs.filter((m) => {
    const level = m.messageLevel || getDefaultLevel(m);
    return level === "progress";
  });

  return (
    <div className={`ml-[19px] pl-3 pb-3 border-l-2 ${guideLine}`}>
      <div className="space-y-2">
        {/* L1 Insights first (highest visual weight) */}
        {insightMsgs.map((msg) => (
          <ResultHighlight key={msg.id} message={msg} />
        ))}

        {/* L2 Progress (lower visual weight) */}
        {progressMsgs.map((msg) => (
          <div key={msg.id} className="py-0.5">
            <span className="text-[11px] text-muted-foreground/50
              leading-relaxed">
              {msg.content}
            </span>
          </div>
        ))}
      </div>
    </div>
  );
}
```

**Constraints:**

- The `ml-[19px]` aligns the guide line with the center of the colored dot above
- L1 insights render BEFORE L2 progress (insights have higher priority)
- Bottom padding `pb-3` creates clear separation from next step

                  agentColors[step.agentRole] || "";
                const isStepExpanded =
                  expandedSteps.has(step.id);
                const msgs = stepMessages[step.id] || [];
                const hasInsight = msgs.some(
                  (m) =>
                    m.messageLevel === "insight"
                    || m.canvasLinkId
                );

                return (
                  <div key={step.id}
                    className="border-b border-border/6
                      last:border-b-0">
                    {/* Step header row */}
                    <button
                      onClick={() =>
                        toggleStepResults(step.id)
                      }
                      className="w-full flex items-center
                        gap-2.5 px-4 py-2.5 text-left
                        hover:bg-muted/5 transition-colors"
                    >
                      <span className={`w-2 h-2 rounded-full
                        shrink-0 ${dotColor}`} />
                      <span className="text-[12px]
                        text-foreground/60 font-medium
                        truncate">
                        {icon} {step.label}
                      </span>
                      <span className="flex-1" />
                      {!isStepExpanded && step.summary && (
                        <span className="text-[11px]
                          text-muted-foreground/30 truncate
                          max-w-[40%]">
                          {step.summary}
                        </span>
                      )}
                      {!isStepExpanded && hasInsight && (
                        <span className={`text-[10px]
                          ${textColor} opacity-40
                          shrink-0`}>
                          ◆
                        </span>
                      )}
                      <span className="text-[10px]
                        text-muted-foreground/25 shrink-0
                        ml-1">
                        {isStepExpanded ? "▾" : "▸"}
                      </span>
                    </button>

                    {/* Expanded: results with guide line */}
                    {isStepExpanded && (
                      <StepExpandedContent
                        step={step}
                        messages={msgs}
                      />
                    )}
                  </div>
                );
              })}
            </div>
          )}
        </div>
      )}
    </div>
  );
}
```
